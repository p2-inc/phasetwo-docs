# --- General crawlers ---------------------------------------------------------
User-agent: *
Allow: /
# Every page is also published as raw markdown at the same URL with ".md" appended.
# Those twins are for tooling to fetch on demand, not for search results, and an
# indexed .md would compete with the HTML page it duplicates. GitHub Pages serves no
# custom headers, so the exclusion has to live here.
Disallow: /*.md$

# Deliberately crawlable: /search, /blog/tags/, /blog/page/, /blog/archive/ and
# /blog/authors/. These are thin taxonomy and utility routes we do not want in
# the index, so each one serves <meta name="robots" content="noindex, follow">.
# A crawler has to fetch a page to see that tag -- disallowing these paths would
# block the very mechanism that keeps them out of the index (and leave any URL
# already indexed stuck there). Please do not re-add a Disallow for them; they
# are also excluded from the sitemap in docusaurus.config.js.
